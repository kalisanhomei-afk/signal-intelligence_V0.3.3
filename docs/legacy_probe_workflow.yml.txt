name: SIGNAL V0.3.2 read-only feed preflight
on:
  workflow_dispatch:
permissions:
  contents: read
jobs:
  preflight:
    runs-on: ubuntu-latest
    timeout-minutes: 8
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: '3.12'
      - name: Offline contracts
        run: |
          python scripts/test_pipeline.py
          python scripts/test_governance.py
          python scripts/test_v032.py
      - name: Live endpoint probe (never approves licenses)
        id: probe
        shell: bash
        run: |
          set +e
          python scripts/probe_sources.py --output data/v032_live_probe.json
          EXIT_CODE=$?
          echo "exit_code=${EXIT_CODE}" >> "$GITHUB_OUTPUT"
          exit 0
      - uses: actions/upload-artifact@v4
        with:
          name: signal-v032-feed-probe
          path: data/v032_live_probe.json
      - name: Flag incomplete live probe
        if: steps.probe.outputs.exit_code != '0'
        run: |
          echo 'Some feeds did not pass. Download artifact for errors; do not approve automatically.'
          exit 1
