name: SIGNAL V0.3 public-source discovery
on:
  workflow_dispatch:
  schedule:
    # UTC 00:10 / 12:10 = 台灣時間 08:10 / 20:10；GitHub 不保證準點
    - cron: '10 0,12 * * *'
permissions:
  contents: write
concurrency:
  group: signal-refresh
  cancel-in-progress: false
jobs:
  update:
    # Default-off: enable only AFTER every active feed has signed-off rights and live checks.
    if: vars.SIGNAL_PRODUCTION_ENABLED == 'true'
    runs-on: ubuntu-latest
    timeout-minutes: 10
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: '3.12'
      - name: Enforce formal source governance for all enabled feeds
        run: python scripts/audit_sources.py --production
      - name: Governance contract tests and source inventory audit
        run: |
          python scripts/test_governance.py
          python scripts/audit_sources.py --output data/source_audit.json
      - name: Offline pipeline tests (no secret required)
        run: python scripts/test_pipeline.py
      - name: Collect public RSS/Atom titles
        id: refresh
        shell: bash
        run: |
          set +e
          python scripts/refresh.py --production --min-success 4
          EXIT_CODE=$?
          echo "exit_code=${EXIT_CODE}" >> "$GITHUB_OUTPUT"
          exit 0
      - name: Save last attempt and source health even on source outage
        shell: bash
        run: |
          git config user.name 'SIGNAL Feed Bot'
          git config user.email '41898282+github-actions[bot]@users.noreply.github.com'
          git add data/discovered.json data/source_audit.json
          if ! git diff --staged --quiet; then
            git commit -m 'data: update discovery and source health'
            git push
          fi
      - name: Fail visibly if every source was unavailable
        if: steps.refresh.outputs.exit_code != '0'
        run: |
          echo 'No feed could be fetched. Last known items were retained; see data/discovered.json for errors.'
          exit 1
